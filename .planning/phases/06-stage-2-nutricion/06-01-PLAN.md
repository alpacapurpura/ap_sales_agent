    class StageCostService:
        """Generic cost service for any stage (replaces stage-specific services)."""
        def __init__(self, db: Session): self.db = db

        def get_channel_costs(self, tenant_id: UUID, stage: str = "nurture") -> Dict[str, float]:
            """Get manual costs for nurture channels from ChannelCostSettingModel.
            Same query as CaptureCostService.get_channel_costs() but can filter by stage
            if a stage column exists, otherwise returns all channel costs."""

        def get_retargeting_spend(self, tenant_id: UUID, start_date, end_date) -> Dict[str, float]:
            """Get retargeting ad spend from MetricAggregationModel for:
            meta-retargeting, google-retargeting, tiktok-retargeting channels.
            Returns: {'meta-retargeting': 450.0, 'google-retargeting': 200.0, ...}"""

        def calculate_cost_per_mql(self, total_costs: float, total_mqls: int) -> Optional[float]:
            """Cost per MQL = total costs / total MQLs. Returns None if mqls == 0."""

        def get_group_cost_per_mql(self, group: str, tenant_id: UUID, start_date, end_date, total_mqls: int) -> Optional[float]:
            """Calculate cost/MQL for a specific group.
            For 'retargeting': sum retargeting ad spend only.
            For 'automation': sum Mailerlite subscription (manual) + AI SDR token costs.
            Returns None if total_mqls == 0.
            This powers the per-group cost breakdown in ChannelGroup headers per CONTEXT.md decision."""

    ```python
    from fastapi import APIRouter, Depends
    from sqlalchemy.orm import Session
    from uuid import UUID
    from src.core.database import get_db

    router = APIRouter(prefix="/webhooks", tags=["Webhooks"])

    @router.post("/mailerlite/{tenant_id}")
    async def handle_mailerlite_webhook(
        tenant_id: UUID,
        payload: dict,
        db: Session = Depends(get_db),
    ):
        """Handle Mailerlite webhook events (campaign.open, campaign.click).
        Creates journey_events and triggers lead score recalculation."""
        event_type = payload.get("type", "")
        subscriber = payload.get("subscriber", {})
        email = subscriber.get("email")
        if not email:
            return {"status": "ignored", "reason": "no_email"}

        # 1. Find customer profile by email
        from src.modules.crm.infrastructure.repositories.customer_repository import CustomerRepository
        customer_repo = CustomerRepository(db)
        profile = customer_repo.find_by_email(tenant_id, email)
        if not profile:
            return {"status": "ignored", "reason": "unknown_subscriber"}

        # 2. Map event type to event_name
        if "open" in event_type:
            event_name = "email_opened"
        elif "click" in event_type:
            event_name = "email_clicked"
        else:
            return {"status": "ignored", "reason": f"unsupported_event: {event_type}"}

        # 3. Create journey_event (use late-binding import)
        from src.modules.crm.infrastructure.models.journey_event_model import JourneyEventModel
        journey_event = JourneyEventModel(
            profile_id=profile.id,
            tenant_id=tenant_id,
            event_name=event_name,
            event_type="track",
            properties={
                "campaign_id": str(payload.get("campaign", {}).get("id", "")),
                "campaign_name": payload.get("campaign", {}).get("name", ""),
                "source": "mailerlite_webhook",
            },
        )
        db.add(journey_event)

        # 4. Recalculate score (triggers MQL transition if threshold crossed)
        from src.modules.crm.application.services.lifecycle_service import LifecycleService
        lifecycle_svc = LifecycleService(db)
        lifecycle_svc.recalculate_score(profile.id, tenant_id)
        db.commit()

        return {"status": "processed", "event": event_name, "profile_id": str(profile.id)}
    ```

    ```python
    async def run_mailerlite_etl_sync(ctx: dict) -> dict:
        """Backup ETL sync: fetch Mailerlite campaign stats every 6 hours.

        Retrieves recent campaign activity from Mailerlite API, finds events
        not yet recorded as journey_events (idempotent via campaign_id + subscriber_email
        deduplication), creates missing journey_events, and triggers score recalculation
        for affected profiles.

        This catches events missed by webhooks (webhook downtime, delivery failures).
        """
        from src.core.database import SessionLocal
        from src.modules.connections.infrastructure.marketing_connectors.mailerlite import MailerLiteConnector
        from src.modules.crm.infrastructure.repositories.customer_repository import CustomerRepository
        from src.modules.crm.infrastructure.models.journey_event_model import JourneyEventModel
        from src.modules.crm.application.services.lifecycle_service import LifecycleService
        from sqlalchemy import select, and_

        logger.info("Starting Mailerlite ETL backup sync")
        db_factory = ctx.get("db_factory")
        if not db_factory:
            logger.error("No db_factory in context")
            return {"status": "error", "reason": "no_db_factory"}

        async with db_factory() as db:
            # 1. Get all tenants with active Mailerlite connections
            # Query ConnectionModel where provider='mailerlite' and status='active'
            from src.modules.connections.infrastructure.models import ConnectionModel
            result = await db.execute(
                select(ConnectionModel).where(
                    and_(
                        ConnectionModel.provider == "mailerlite",
                        ConnectionModel.status == "active",
                    )
                )
            )
            connections = result.scalars().all()

            synced_count = 0
            for conn in connections:
                tenant_id = conn.tenant_id
                try:
                    # 2. Initialize Mailerlite connector with tenant credentials
                    connector = MailerLiteConnector(api_key=conn.credentials.get("api_key"))

                    # 3. Fetch recent campaign activity (last 7 hours to overlap with 6h interval)
                    # Use Mailerlite API: GET /campaigns with filter[status]=sent
                    # For each campaign, get subscriber activity (opens, clicks)
                    # NOTE: Exact API calls depend on MailerLiteConnector methods. If connector
                    # doesn't have campaign activity methods yet, add TODO and log warning.
                    if not hasattr(connector, "get_recent_campaign_activity"):
                        logger.warning(
                            "MailerLiteConnector.get_recent_campaign_activity() not implemented yet. "
                            "Skipping tenant %s. Add method to connector when Mailerlite API integration is complete.",
                            tenant_id,
                        )
                        continue

                    activities = await connector.get_recent_campaign_activity(hours=7)

                    # 4. For each activity, check if journey_event already exists
                    customer_repo = CustomerRepository(db)
                    lifecycle_svc = LifecycleService(db)

                    for activity in activities:
                        email = activity.get("email")
                        campaign_id = activity.get("campaign_id")
                        event_type = activity.get("event_type")  # "open" or "click"

                        profile = customer_repo.find_by_email(tenant_id, email)
                        if not profile:
                            continue

                        event_name = "email_opened" if event_type == "open" else "email_clicked"

                        # Dedup: check if this exact event already exists
                        existing = await db.execute(
                            select(JourneyEventModel).where(
                                and_(
                                    JourneyEventModel.profile_id == profile.id,
                                    JourneyEventModel.tenant_id == tenant_id,
                                    JourneyEventModel.event_name == event_name,
                                    JourneyEventModel.properties["campaign_id"].astext == str(campaign_id),
                                )
                            )
                        )
                        if existing.scalar_one_or_none():
                            continue  # Already recorded

                        # 5. Create missing journey_event
                        journey_event = JourneyEventModel(
                            profile_id=profile.id,
                            tenant_id=tenant_id,
                            event_name=event_name,
                            event_type="track",
                            properties={
                                "campaign_id": str(campaign_id),
                                "campaign_name": activity.get("campaign_name", ""),
                                "source": "mailerlite_etl_sync",
                            },
                        )
                        db.add(journey_event)

                        # 6. Recalculate score for affected profile
                        lifecycle_svc.recalculate_score(profile.id, tenant_id)
                        synced_count += 1

                    await db.commit()
                except Exception as e:
                    logger.error("Mailerlite ETL sync failed for tenant %s: %s", tenant_id, e)
                    await db.rollback()

        logger.info("Mailerlite ETL backup sync complete: %d events synced", synced_count)
        return {"status": "ok", "synced": synced_count}
    ```
