import { useMutation, useQueryClient } from '@tanstack/react-query';

export function usePromoteEvangelist() {

function formatDualCurrency(amount: number, currency: string, usdAmount: number | null): string {
  const fmt = new Intl.NumberFormat('es-MX', { style: 'currency', currency, minimumFractionDigits: 0, maximumFractionDigits: 0 });
  const main = fmt.format(amount);
  if (currency === 'USD') return main;
  if (usdAmount != null) {

    return `${main} (~${usdFmt.format(usdAmount)} USD)`;

  return main;
}

export function NpsSummaryCard({ nps, ugcCount, ugcWritten, ugcAudio }: Props) {
  // NPS Score display: text-xl font-semibold
  // Proportional bar using CSS widths (same technique as HealthBar.tsx):
  //   - Calculate total = promoterCount + passiveCount + detractorCount
  //   - Promoter width %: Math.max(promoterCount/total*100, promoterCount > 0 ? 1 : 0)
  //   - Passive width %: Math.max(passiveCount/total*100, passiveCount > 0 ? 1 : 0)
  //   - Detractor width %: Math.max(detractorCount/total*100, detractorCount > 0 ? 1 : 0)
  //   - Colors: bg-emerald-500 (promoter), bg-yellow-500 (passive), bg-red-500 (detractor)
  //   - Container: flex h-3 rounded-full overflow-hidden
