Methods:
```python
class OpportunityMetricsRepository:
    def __init__(self, db: Session):
        self.db = db

    def count_new_sqls(self, tenant_id: UUID, start_date: datetime, end_date: datetime) -> int:
        """Count profiles that transitioned TO SQL stage in period."""
        stmt = select(func.count(func.distinct(LifecycleTransitionModel.profile_id))).where(
            LifecycleTransitionModel.tenant_id == tenant_id,
            LifecycleTransitionModel.to_stage == "SQL",
            LifecycleTransitionModel.occurred_at >= start_date,
            LifecycleTransitionModel.occurred_at <= end_date,
        )
        return int(self.db.execute(stmt).scalar() or 0)

    def count_mqls_in_period(self, tenant_id: UUID, start_date: datetime, end_date: datetime) -> int:
        """Count profiles that transitioned TO MQL stage (for mini funnel source)."""
        # Same as NurtureMetricsRepository.count_new_mqls
        stmt = select(func.count(func.distinct(LifecycleTransitionModel.profile_id))).where(
            LifecycleTransitionModel.tenant_id == tenant_id,
            LifecycleTransitionModel.to_stage == "MQL",
            LifecycleTransitionModel.occurred_at >= start_date,
            LifecycleTransitionModel.occurred_at <= end_date,
        )
        return int(self.db.execute(stmt).scalar() or 0)

    def count_checkout_events(self, tenant_id: UUID, start_date: datetime, end_date: datetime) -> dict:
        """Count checkout_initiated and cart_abandoned journey_events.
        Returns: {"checkout_initiated": int, "cart_abandoned": int, "checkout_value": float, "abandoned_value": float}
        """
        event_names = ["checkout_initiated", "cart_abandoned"]
        stmt = select(
            JourneyEventModel.event_name,
            func.count(JourneyEventModel.id),
            func.coalesce(func.sum(
                func.cast(JourneyEventModel.properties["total_price"].as_string(), Float)
            ), 0.0),
        ).where(
            JourneyEventModel.tenant_id == tenant_id,
            JourneyEventModel.event_name.in_(event_names),
            JourneyEventModel.created_at >= start_date,
            JourneyEventModel.created_at <= end_date,
        ).group_by(JourneyEventModel.event_name)
        results = self.db.execute(stmt).all()
        counts = {name: {"count": 0, "value": 0.0} for name in event_names}
        for event_name, count, value in results:
            counts[event_name] = {"count": count, "value": float(value)}
        return counts

    def count_meeting_events(self, tenant_id: UUID, start_date: datetime, end_date: datetime) -> dict:
        """Count meeting_booked, meeting_completed, meeting_no_show, meeting_rescheduled.
        Returns: {"booked": int, "completed": int, "no_show": int, "rescheduled": int}
        """
        event_names = ["meeting_booked", "meeting_completed", "meeting_no_show", "meeting_rescheduled"]
        stmt = select(
            JourneyEventModel.event_name,
            func.count(JourneyEventModel.id),
        ).where(
            JourneyEventModel.tenant_id == tenant_id,
            JourneyEventModel.event_name.in_(event_names),
            JourneyEventModel.created_at >= start_date,
            JourneyEventModel.created_at <= end_date,
        ).group_by(JourneyEventModel.event_name)
        results = self.db.execute(stmt).all()
        counts = {"booked": 0, "completed": 0, "no_show": 0, "rescheduled": 0}
        name_map = {"meeting_booked": "booked", "meeting_completed": "completed", "meeting_no_show": "no_show", "meeting_rescheduled": "rescheduled"}
        for event_name, count in results:
            key = name_map.get(event_name)
            if key:
                counts[key] = count
        return counts

    def count_payment_link_events(self, tenant_id: UUID, start_date: datetime, end_date: datetime) -> dict:
        """Count payment_link_sent journey_events.
        Returns: {"count": int, "value": float}
        """
        stmt = select(
            func.count(JourneyEventModel.id),
            func.coalesce(func.sum(
                func.cast(JourneyEventModel.properties["total_price"].as_string(), Float)
            ), 0.0),
        ).where(
            JourneyEventModel.tenant_id == tenant_id,
            JourneyEventModel.event_name == "payment_link_sent",
            JourneyEventModel.created_at >= start_date,
            JourneyEventModel.created_at <= end_date,
        )
        result = self.db.execute(stmt).one()
        return {"count": result[0] or 0, "value": float(result[1] or 0.0)}
```
