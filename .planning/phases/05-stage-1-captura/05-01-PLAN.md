```python
class CaptureMetricsRepository:
    def __init__(self, db: Session):
        self.db = db

    def count_leads_by_source(self, tenant_id: UUID, start_date: datetime, end_date: datetime) -> Dict[str, int]:
        """Count net-new profiles by lead_source within date range.
        Returns: {"ig-dm": 45, "landing-form": 120, ...}
        """
        # SQLAlchemy 2.0 select() syntax
        stmt = (
            select(
                CustomerProfileModel.lead_source,
                func.count(CustomerProfileModel.id).label("lead_count"),
            )
            .where(
                CustomerProfileModel.tenant_id == tenant_id,
                CustomerProfileModel.first_seen_at >= start_date,
                CustomerProfileModel.first_seen_at <= end_date,
                CustomerProfileModel.lead_source.isnot(None),
                CustomerProfileModel.deleted_at.is_(None),  # soft delete
            )
            .group_by(CustomerProfileModel.lead_source)
        )
        rows = self.db.execute(stmt).all()
        return {row.lead_source: row.lead_count for row in rows}

    def count_conversations_by_channel(self, tenant_id: UUID, start_date: datetime, end_date: datetime) -> Dict[str, int]:
        """Count distinct conversation sessions per messaging channel for AI Agent secondary line.
        Query journey_events where event_name='message_received' grouped by channel.
        Returns: {"ig-dm": 140, "whatsapp-inbound": 230, ...}
        """
        # Query JourneyEventModel for message interactions grouped by properties->channel_slug
        # Use JSON extraction for properties.channel_slug
        from src.modules.crm.infrastructure.models.customer_model import JourneyEventModel
        stmt = (
            select(
                func.json_extract_path_text(JourneyEventModel.properties, "channel_slug").label("channel"),
                func.count(func.distinct(JourneyEventModel.session_id)).label("conv_count"),
            )
            .where(
                JourneyEventModel.tenant_id == tenant_id,
                JourneyEventModel.event_name == "message_received",
                JourneyEventModel.created_at >= start_date,
                JourneyEventModel.created_at <= end_date,
            )
            .group_by("channel")
        )
        rows = self.db.execute(stmt).all()
        return {row.channel: row.conv_count for row in rows if row.channel}
```

```python
class CaptureCostService:
    def __init__(self, db: Session):
        self.db = db

    def get_channel_costs(self, tenant_id: UUID, stage: str = "capture") -> Dict[str, float]:
        """Get monthly cost per channel slug from channel_cost_settings.
        Returns: {"mailerlite": 29.0, "whatsapp-inbound": 15.0, ...}
        Sums multiple cost entries per channel (platform + agency + tool).
        """

    def get_total_stage0_spend(self, tenant_id: UUID, start_date: datetime, end_date: datetime) -> float:
        """Get total Stage 0 paid channel spend from metric_aggregations table.
        Reads spend metric for meta-ads, google-ads, tiktok-ads, yt-ads channels.
        """

    def calculate_cal(self, total_costs: float, total_leads: int) -> Optional[float]:
        """CAL = total costs / total leads. Returns None if leads == 0."""
        if total_leads == 0:
            return None
        return round(total_costs / total_leads, 2)

    def get_prorated_agency_costs(self, tenant_id: UUID, connected_channels: List[str]) -> Dict[str, float]:
        """Distribute agency costs across channels based on proration_category:
        - organic_management: evenly across connected organic social channels (ig-dm, fb-messenger, tiktok-dm)
        - paid_management: proportional to ad spend per paid channel
        - video: evenly across video-capable channels (ig-dm, tiktok-dm)
        - full_service: evenly across all connected channels
        """
```
